

各组件的扩展
```
Accepter：只专门负责处理服务端的连接，所以一般不需要对其进行扩展
Channel：若有新事件类型的扩展，则更新其回调函数成员以及事件分发函数即可
TcpConnection：如果要扩展客户端的新业务，只需在类中添加新回调，并通过构造函数进行注册即可
与具体数据有关的业务扩展：根据解析的数据调用不同方法，只需在TcpConnection中添加新业务，然后在读回调中建立中建立分支并调用即可
```

相关细节
```
目前Reactor为单线程版，如果在分发事件的分支中直接处理业务会影响整个主线程的运行(处理业务要时间，特别是数据库查询，会影响其他客户端事件的处理)，暂时不关心
太多，直接在读事件回调中建立新的回调分支，并直接调用即可，后续的多线程版中(主从Reactor)，将具体业务回调移交给其他线程完成，主线程只负责监听连接与事件分发
```

```c++
#include "TcpConnection.h"

TcpConnection::TcpConnection(int fd,Epoll* ep):ep_(ep),sock_(fd),ch_(fd,ep)
{    
    ch_.EnableReading();
    ch_.SetET();
    
    ch_.SetErrorCallBack( [this](){ HandleClose(); } );
    ch_.SetReadCallBack( [this](){ HandleRead(); } );
    ch_.SetWriteCallBack( [this](){ HandleWrite(); } );

    ep->Update(&ch_);
}

void TcpConnection::SetCloseCallBack(std::function<void(int)> fun){ closecallback_=fun; }

TcpConnection::~TcpConnection(){}

void TcpConnection::HandleClose()
{
    if(ch_.inepoll())
        ep_->Del(&ch_);
    else 
        return;
    
    if(closecallback_) closecallback_(sock_.fd());
}

    while(true)
    {
        size_t readables=buffer_.CheckWritAble(1024);
        int nums=recv(sock_.fd(),buffer_.WritePtr(),readables,0);

        if(nums<0)
        {
            if(errno==EAGAIN) break;
            std::cout<<"call recv failed"<<std::endl;
            HandleClose();
            return;
        }
        else if(nums==0)
        {
            HandleClose();
            return;
        }
        else
        {
            buffer_.MoveWritePtr(nums);
```

客户端(简单测试)
```c++
#include<sys/socket.h>
#include<iostream>
#include<unistd.h>
#include<arpa/inet.h>
#include<memory.h>
#include<string.h>
#include<mutex>
#include<thread>
bool g_exit=false;//用于通知子线程退出的标志
std::mutex check_exit;//为退出标志加锁的处理

void InitSocket(int& socket_fd,sockaddr_in& server_sai)
{
    socket_fd=socket(AF_INET,SOCK_STREAM,0);
    if(socket_fd==-1)
        throw std::runtime_error("create socket failed!");
    
    server_sai.sin_family=AF_INET;
    server_sai.sin_port=htons(2026);
    inet_aton("172.20.10.2",&(server_sai.sin_addr));
}

void Recv(int socket_fd)//子线程读取函数
{
    while(true)
    {
        {
            std::lock_guard<std::mutex> lg1(check_exit);
            if(g_exit) break;
        }
        char buffer[64]={0};
        int nums=recv(socket_fd,buffer,sizeof(buffer),MSG_DONTWAIT);
        if(nums>0)
            std::cout<<"server message:"<<buffer<<std::endl;
    }
}

void Connect(int& socket_fd,sockaddr_in& server_sai)
{
    if(connect(socket_fd,(sockaddr*)&server_sai,sizeof(server_sai))!=-1)
        std::cout<<"connect success!"<<std::endl;
    else
    {
        std::cout<<"connect error!"<<std::endl;
        return;
    }

    std::thread recv_thread(Recv,socket_fd);

    char buffer[64];
    int write_nums;
    while(memset(buffer,0,sizeof(buffer)),true)
    {
        std::cin>>buffer;
        if(!strcmp(buffer,"exit"))
        {
            std::lock_guard<std::mutex> lg1(check_exit);
            g_exit=true;
            break;
        }

        int len=strlen(buffer);
        int head=htonl(len);
        send(socket_fd,&head,sizeof(head),0);
        write_nums=send(socket_fd,buffer,len,0);
        std::cout<<"current write nums:"<<write_nums<<std::endl;
    }

    close(socket_fd);
    recv_thread.join();
}

int main()
{
    int fd;
    sockaddr_in server_sai;
    InitSocket(fd,server_sai);
    Connect(fd,server_sai);

    return 0;
}
```
