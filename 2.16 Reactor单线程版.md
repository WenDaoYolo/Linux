```c++
#include "RecvBuffer.h"

RecvBuffer::RecvBuffer(int incSize):space_(incSize),readIndex_(0),writeIndex_(0){}

size_t RecvBuffer::ReadAbleBytes(){ return writeIndex_-readIndex_; }

const char* RecvBuffer::ReadPtr(){ return &(space_[readIndex_]); }

char* RecvBuffer::WritePtr(){ return &(space_[writeIndex_]); }

void RecvBuffer::MoveWritePtr(size_t offset){ writeIndex_+=offset; }

void RecvBuffer::MoveReadPrt(size_t offset)
{
    readIndex_+=offset;
    if(readIndex_==writeIndex_) readIndex_=writeIndex_=0;
}

size_t RecvBuffer::CheckWritAble(size_t len)
{
    if(space_.size()-writeIndex_<len)
    {
        memmove(&space_[0],ReadPtr(),ReadAbleBytes());
        writeIndex_-=readIndex_;
        readIndex_=0;

        if(space_.size()-writeIndex_<len)
            space_.resize(space_.size()+len*2);
    }
    return space_.size()-writeIndex_;
}

RecvBuffer::~RecvBuffer(){};
```

```c++
#include "Socket.h"

int CreateSocket(int flag)
{
    int fd=-1;
    if(flag)
        fd=socket(AF_INET,SOCK_STREAM,0);
    else
        fd=socket(AF_INET,SOCK_STREAM|SOCK_NONBLOCK,0);

    if(fd==-1) std::cout<<"CreateSocket failed"<<std::endl;
    return fd;
}

Socket::Socket(int fd):fd_(fd){}
Socket::~Socket(){ close(fd_); }

void Socket::SetBlockTime(time_t sec,time_t usec)
{
    timeval tv={sec,usec};
    setsockopt(fd_,SOL_SOCKET,SO_RCVTIMEO,&tv,sizeof(tv));
}

void Socket::SetNonBlock()
{
    int state=fcntl(fd_,F_GETFL,NULL);
    if(state==-1) state=0;
    fcntl(fd_,F_SETFL,state|SOCK_NONBLOCK);
}

void Socket::SetReuseAddr(bool flag)
{
    int res=flag;
    setsockopt(fd_,SOL_SOCKET,SO_REUSEADDR,&res,sizeof(res));
}

void Socket::SetReusePort(bool flag)
{
    int res=flag;
    setsockopt(fd_,SOL_SOCKET,SO_REUSEPORT,&res,sizeof(res));
}

void Socket::SetKeepAlive(bool flag)
{
    int res=flag;
    setsockopt(fd_,SOL_SOCKET,SO_KEEPALIVE,&res,sizeof(res));
}

void Socket::SetTcpNoDelay(bool flag)
{
    int res=flag;
    setsockopt(fd_,SOL_SOCKET,TCP_NODELAY,&res,sizeof(res));
}

void Socket::bind(InetAddress IA){ ::bind(fd_,IA.addr(),sizeof(sockaddr)); }

void Socket::listen(int queue_len){ ::listen(fd_,queue_len); }

int Socket::accept(InetAddress& IA)
{
    sockaddr_in sai;
    socklen_t len=sizeof(sai);
    int client_fd=::accept(fd_,(sockaddr*)&sai,&len);

    IA.SetAddrIn(sai);
    if(client_fd==-1) std::runtime_error("Socket::accept failed");
    return client_fd;
}

int Socket::accept4(InetAddress& IA)
{
    sockaddr_in sai;
    socklen_t len=sizeof(sai);
    int client_fd=::accept4(fd_,(sockaddr*)&sai,&len,SOCK_NONBLOCK|SOCK_CLOEXEC);

    IA.SetAddrIn(sai);
    if(client_fd==-1) std::runtime_error("Socket::accept failed");
    return client_fd;
}

const int Socket::fd(){ return fd_; }
```

```c++
        ~Epoll();                                        //析构函数(关闭epollfd)

        void AddFd(int fd,int op);                       //将socket添加到红黑树
        std::vector<epoll_event> loop(int timeout=-1);   //监听并获取发生的事件
};
```

Epoll.cpp
```c++
#include "Epoll.h"

Epoll::Epoll()
{
    epfd_=epoll_create(1);
    if(epfd_==-1) std::cout<<"epoll_create faild"<<std::endl;
}

Epoll::~Epoll(){ close(epfd_); }

void Epoll::AddFd(int fd,int op)
{
    epoll_event ev;
    ev.data.fd=fd;
    ev.events=op;

    if(epoll_ctl(epfd_,EPOLL_CTL_ADD,fd,&ev)==-1)
        std::cout<<"epoll_ctl ADD failed"<<std::endl;
}

std::vector<epoll_event> Epoll::loop(int timeout)
{
    int events=epoll_wait(epfd_,revents_,maxEvents_,timeout);
    std::vector<epoll_event> res;

    if(events<0)
    {
        if(errno!=EINTR)
            std::cout<<"call epoll_wait failed"<<std::endl;
    }
    else if(events==0)
    {
        std::cout<<"no event"<<std::endl;
    }
    else
    {
        for(int i=0;i<events;i++)
            res.push_back(revents_[i]);
    }
    return res;
}

void Epoll::DelFd(int fd)
{
    if(epoll_ctl(epfd_,EPOLL_CTL_DEL,fd,NULL)==-1)
        std::cout<<"epoll_ctl EDL failed"<<std::endl;
}
```

```c++
void Channel::SetWriteCallBack(std::function<void()> fun){ writecallback_=fun; }
void Channel::SetErrorCallBack(std::function<void()> fun){ errorcallback_=fun; }

void Channel::HandleEvent()
{   
    if((revents_&EPOLLERR)||(revents_&EPOLLHUP))//注意，一个socket可能同时发生多种事件，所以需要独立的判断分支
    {
        if(errorcallback_) errorcallback_();    //如果不为空则执行，避免访问空指针
    }
    if(revents_&EPOLLIN)
    {
        if(readcallback_) readcallback_();
    }
    if(revents_&EPOLLOUT)
    {
        if(writecallback_) writecallback_();
    }
}
```

Acceptor.cpp
```c++
#include "Accepter.h"

Accepter::Accepter(std::string ip,uint16_t port,Epoll* ep):sock_(CreateSocket(0)),ch_(sock_.fd(),ep),ep_(ep)
{
    sock_.SetReuseAddr(true);
    sock_.SetReusePort(true);

    InetAddress listenIA(ip,port);
    sock_.bind(listenIA);
    sock_.listen(10);
    
    ch_.EnableReading();
    ch_.SetET();
    ch_.SetReadCallBack//将处理新连接的回调注册到通道的读事件中
    (
        [this]()
        {
            InetAddress IA;
            int fd=sock_.accept4(IA);//闭包中创建fd和IA后，再作为参数，通过成员调用外部定义的可调用对象
            if(fd!=-1&&newconnectioncallback_)
                newconnectioncallback_(fd,IA);
        }
    );

    ep_->Update(&ch_);//将通道及其socket添加到红黑树
}

void Accepter::SetNewConnectionCallBack(std::function<void(int)> fun){ newconnectioncallback_=fun; }

Accepter::~Accepter()
{
    ep_->Del(&ch_);//将socket从红黑树中移除(监听socket)
}
```

```c++
#include "TcpConnection.h"

TcpConnection::TcpConnection(int fd,Epoll* ep):ep_(ep),sock_(fd),ch_(fd,ep)
{    
    ch_.EnableReading();
    ch_.SetET();
    
    ch_.SetErrorCallBack( [this](){ HandleClose(); } );
    ch_.SetReadCallBack( [this](){ HandleRead(); } );
    ch_.SetWriteCallBack( [this](){ HandleWrite(); } );

    ep->Update(&ch_);
}

void TcpConnection::SetCloseCallBack(std::function<void(int)> fun){ closecallback_=fun; }

TcpConnection::~TcpConnection(){}

void TcpConnection::HandleClose()
{
    if(ch_.inepoll())
        ep_->Del(&ch_);
    else 
        return;//如果不在红黑树则直接退出，防止重复调用关闭连接的回调
    
    if(closecallback_) closecallback_(sock_.fd());//关闭后该内存会销毁，不要再执行仍和访问操作
}

void TcpConnection::HandleRead()
{
    while(true)
    {
        size_t readables=buffer_.CheckWritAble(1024);
        int nums=recv(sock_.fd(),buffer_.WritePtr(),readables,0);

        if(nums<0)
        {
            if(errno==EAGAIN) break;
            std::cout<<"call recv failed"<<std::endl;
            HandleClose();
            return;//同理内存会销毁，直接返回，不要再执行任何访问操作
        }
        else if(nums==0)
        {
            HandleClose();
            return;//同理内存会销毁，直接返回，不要再执行任何访问操作
        }
        else
        {
            buffer_.MoveWritePtr(nums);
        }
    }

    while(buffer_.ReadAbleBytes()>=4)
    {
        int len=0;
        memcpy(&len,buffer_.ReadPtr(),sizeof(int));
        len=ntohl(len);
        
        if(buffer_.ReadAbleBytes()>=4+len)
        {
            buffer_.MoveReadPrt(4);
            std::string msg(buffer_.ReadPtr(),len);
            buffer_.MoveReadPrt(len);
            std::cout<<"client"<<sock_.fd()<<":"<<msg<<std::endl;
        }
        else
            break;//如果数据不完整则跳出循环，等待下次解析
    }
}
```

```c++
#include "Accepter.h"
#include "TcpConnection.h"
#include<unordered_map>

std::unordered_map<int,TcpConnection*> g_maps;
Epoll g_ep;

void Loop()
{
    Accepter acpt("172.20.10.2",2026,&g_ep);
    acpt.SetNewConnectionCallBack
    (
        [&](int client_fd,InetAddress IA)
        {
            TcpConnection* tmp=new TcpConnection(client_fd,&g_ep);
            g_maps[client_fd]=tmp;

            tmp->SetCloseCallBack
            (
                [&](int close_fd)
                {
	                //注意捕获变量的生命周期，不能使用外部闭包的局部变量
                    TcpConnection* del=g_maps[close_fd];
                    g_maps.erase(close_fd);
                    delete del;
                    std::cout<<"client"<<close_fd<<" disconnect"<<std::endl;
                }
            );
            std::cout<<"client "<<client_fd<<" connect"<<std::endl;
            std::cout<<"IP:"<<IA.ip()<<" PORT:"<<IA.port()<<std::endl;
        }
    );

    while(true)
    {
        std::cout<<"waiting events..."<<std::endl;
	    std::vector<Channel*> channels=g_ep.loop();

        for(int i=0;i<channels.size();i++)
        {
            channels[i]->HandleEvent();
        }
    }
}

int main()
{
	Loop();

	return 0;
}
```
